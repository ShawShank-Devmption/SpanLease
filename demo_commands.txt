CURRENT_DEMO_MERMAID

cd /Users/shashankbhat/Projects/os_project

# 1. Select the required Java 21 runtime.
export JAVA_HOME=/opt/homebrew/opt/openjdk@21/libexec/openjdk.jdk/Contents/Home
export PATH="$JAVA_HOME/bin:$PATH"
java -version
command -v jq

# 2. Optional: verify the real grpc-java gate/lifecycle feasibility path.
# This is a real loopback RPC test, but it is not the complete three-service deadlock.
./gradlew :testbed:test --tests dev.spanlease.testbed.GrpcGateFeasibilityTest

# 3. Run the deterministic production-analyzer demo and save its output.
mkdir -p build/demo
./gradlew -q :analyzer:productionDemo | tee build/demo/production-demo.txt

# 4. Extract the closed-ring and spare-capacity JSON reports.
sed -n '2p' build/demo/production-demo.txt | jq . > build/demo/deadlock-report.json
sed -n '4p' build/demo/production-demo.txt | jq . > build/demo/spare-capacity-report.json

# 5. See the closed wait that the analyzer evaluated.
jq '{
  blocked_executions: [.core.executions[] | {
    execution: .id,
    waits_for_invocation: .blocked_on_invocation
  }],
  queued_requests: .core.queue_entries,
  occupied_slots: [.core.units[] | {
    slot: .id,
    owned_by_execution: .owner_execution_id
  }],
  persistence: .persistence
}' build/demo/deadlock-report.json

# 6. Show and assert SpanLease's verdict.
jq -r '"SpanLease verdict: \(.verdict)",
       "Certified persistence: \(.persistence.narrowed_span_ms) ms (required: \(.persistence.tau_ms) ms)"' \
  build/demo/deadlock-report.json
test "$(jq -r '.verdict' build/demo/deadlock-report.json)" = "CONFIRMED_DEADLOCK"
echo "PASS: SpanLease reported CONFIRMED_DEADLOCK"

# 7. Print the evidence graph used by the report.
jq -r '"SpanLease evidence graph:",
       (.core.edges[] | "  \(.from) --\(.kind)--> \(.to)")' \
  build/demo/deadlock-report.json | tee build/demo/deadlock-graph.txt

# 8. Save the same evidence graph in Graphviz DOT format.
jq -r '"digraph SpanLeaseEvidence {",
       "  rankdir=LR;",
       (.core.edges[] |
         "  \"" + .from + "\" -> \"" + .to + "\" [label=\"" + .kind + "\"];") ,
       "}"' build/demo/deadlock-report.json | tee build/demo/deadlock-graph.dot

# 9. Optional: if Graphviz is installed, render and open the graph as SVG.
if command -v dot >/dev/null 2>&1; then
  dot -Tsvg build/demo/deadlock-graph.dot -o build/demo/deadlock-graph.svg
  open build/demo/deadlock-graph.svg
else
  echo "Graphviz is not installed; the text and DOT graphs are still available in build/demo/."
fi

# 10. Show the spare-capacity negative control.
jq -r '"Negative-control verdict: \(.verdict)",
       "Residual evidence nodes: \((.core.executions + .core.queue_entries + .core.units) | length)"' \
  build/demo/spare-capacity-report.json
test "$(jq -r '.verdict' build/demo/spare-capacity-report.json)" = "CONFIRMED_NO_DEADLOCK"
echo "PASS: spare capacity reduced the candidate graph to empty"
